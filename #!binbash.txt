#!/bin/bash
# firstly you to add lun sdd size 3gb then use this script
# chnage the date daily
mkdir /task5
mkfs -t ext4 /dev/sdd
mount -t ext4 /dev/sdd /task5
echo "/dev/sdd  /task5 ext4    defaults        0       0" >> /etc/fstab

for i in {1..14}
do
dd if=/dev/zero of=/task5/swapf$i bs=1M count=130
mkswap /task5/swapf$i
swapon /task5/swapf$i
echo "/task5/swapf$i   swap    swap    defaults        0       0" >> /etc/fstab
done
